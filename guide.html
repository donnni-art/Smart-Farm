▼

💡GPIO34 และ 35 เป็น Input-only (ADC) เหมาะกับ Analog Sensor โดยเฉพาะ ไม่สามารถใช้เป็น Output ได้
2
Soil Moisture Sensor × 2
ต่อเซนเซอร์ความชื้นดิน Zone 1 และ Zone 2
▼
Sensor Pin→ESP32 Pin
VCC (Zone 1)→3.3V
GND (Zone 1)→GND
AOUT (Zone 1)→GPIO34
VCC (Zone 2)→3.3V
GND (Zone 2)→GND
AOUT (Zone 2)→GPIO35
⚠️ใช้ไฟ 3.3V เท่านั้น — ถ้าต่อ 5V จะทำให้ ADC ของ ESP32 เสียหายเพราะ GPIO34/35 รับแรงดันสูงสุด 3.3V
💡ค่า ADC 0 = ชื้นมาก, 4095 = แห้งมาก — ต้อง Calibrate ค่าตามเซนเซอร์แต่ละรุ่น
3
DHT22 — อุณหภูมิ + ความชื้นอากาศ
ต่อ Pull-up Resistor 10kΩ ที่สาย DATA
▼
DHT22 Pin→ESP32
Pin 1 — VCC→3.3V
Pin 2 — DATA→GPIO4
Pin 2 ยังต่อกับ 3.3V ผ่าน 10kΩ
Pin 3 — NC—ไม่ต่อ
Pin 4 — GND→GND
⚠️ต้องต่อ Resistor 10kΩ ระหว่าง DATA และ 3.3V (Pull-up) หากไม่ต่อจะอ่านค่าผิดพลาดหรือไม่ได้ค่าเลย
📦Library Arduino: DHT sensor library โดย Adafruit — ติดตั้งผ่าน Library Manager
4
DS18B20 — เซนเซอร์อุณหภูมิดิน
Waterproof probe, Pull-up 4.7kΩ, โปรโตคอล 1-Wire
▼
สายเคเบิล→ESP32
สายแดง — VCC→3.3V
สายดำ — GND→GND
สายเหลือง — DATA→GPIO5
DATA ต่อ Pull-up 4.7kΩ ไปยัง 3.3V
💡DS18B20 ใช้ 1-Wire protocol — ต่อหลายตัวบน GPIO เดียวกันได้ โดยแยกกัน Address 64-bit
📦Library: DallasTemperature + OneWire by milesburton — ติดตั้งใน Library Manager
5
Relay Module + Solenoid Valve × 4
ควบคุมวาล์ว 12V จาก GPIO 5V — ต้องใช้ไฟ 12V แยก
▼
Relay Pin→ESP32/12V
VCC→5V
GND→GND
IN1→GPIO26
IN2→GPIO27
IN3→GPIO14
IN4→GPIO12
COM+NO (×4)→12V + Valve
⚠️แยกกราวด์! วงจร 12V และ ESP32 (5V/3.3V) ต้องมี Common GND เดียวกัน แต่ต้องแยกสายไฟ AC/DC ออกจากกัน
💡ต่อ Relay แบบ Active-LOW: IN=LOW → Relay เปิด, IN=HIGH → Relay ปิด — ค่า default ใน code คือ LOW = เปิดวาล์ว
โค้ด ESP32 (Arduino Framework)
C++ / Arduino 
// Smart Farm IoT — ESP32 Firmware
// Libraries ที่ต้องติดตั้ง: DHT, DallasTemperature, OneWire, PubSubClient

#include <WiFi.h>
#include <PubSubClient.h>
#include <DHT.h>
#include <OneWire.h>
#include <DallasTemperature.h>

// ── WiFi / MQTT ──────────────────────────────
const char* SSID       = "Your_WiFi_Name";
const char* PASSWORD   = "Your_WiFi_Password";
const char* MQTT_HOST  = "your-mqtt-broker.com";
const int   MQTT_PORT  = 1883;

// ── GPIO Pin Definitions ──────────────────────
#define SOIL1_PIN   34    // Soil Sensor Zone 1 (ADC)
#define SOIL2_PIN   35    // Soil Sensor Zone 2 (ADC)
#define DHT_PIN     4     // DHT22 Data
#define DS18B20_PIN 5     // DS18B20 Data (1-Wire)
#define VALVE1_PIN  26    // Relay IN1 → Valve 1
#define VALVE2_PIN  27    // Relay IN2 → Valve 2
#define VALVE3_PIN  14    // Relay IN3 → Valve 3
#define VALVE4_PIN  12    // Relay IN4 → Valve 4

// ── Sensor Objects ────────────────────────────
DHT dht(DHT_PIN, DHT22);
OneWire oneWire(DS18B20_PIN);
DallasTemperature ds18b20(&oneWire);
WiFiClient wifiClient;
PubSubClient mqtt(wifiClient);

void setup() {
  Serial.begin(115200);

  // ตั้งค่า Relay pins (Active-LOW)
  int valvePins[] = {VALVE1_PIN, VALVE2_PIN, VALVE3_PIN, VALVE4_PIN};
  for (int i = 0; i < 4; i++) {
    pinMode(valvePins[i], OUTPUT);
    digitalWrite(valvePins[i], HIGH);  // ปิดวาล์วทั้งหมดก่อน
  }

  dht.begin();
  ds18b20.begin();

  // เชื่อมต่อ WiFi
  WiFi.begin(SSID, PASSWORD);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("WiFi connected: " + WiFi.localIP().toString());

  mqtt.setServer(MQTT_HOST, MQTT_PORT);
  mqtt.setCallback(mqttCallback);
}

void loop() {
  if (!mqtt.connected()) reconnectMQTT();
  mqtt.loop();

  // อ่านเซนเซอร์ทุก 3 วินาที
  static unsigned long lastRead = 0;
  if (millis() - lastRead > 3000) {
    lastRead = millis();
    readAndPublish();
  }
}

void readAndPublish() {
  // อ่านค่าเซนเซอร์
  int   raw1  = analogRead(SOIL1_PIN);
  int   raw2  = analogRead(SOIL2_PIN);
  float soil1 = map(raw1, 4095, 1500, 0, 100);  // Calibrate ตามเซนเซอร์
  float soil2 = map(raw2, 4095, 1500, 0, 100);
  float temp  = dht.readTemperature();
  float humid = dht.readHumidity();
  ds18b20.requestTemperatures();
  float stemp = ds18b20.getTempCByIndex(0);

  // Publish MQTT (JSON)
  char payload[200];
  snprintf(payload, sizeof(payload),
    "{\"soil1\":%.1f,\"soil2\":%.1f,\"temp\":%.1f,\"humid\":%.1f,\"stemp\":%.1f}",
    soil1, soil2, temp, humid, stemp);
  mqtt.publish("farm/sensors", payload);
}

void mqttCallback(char* topic, byte* payload, unsigned int len) {
  // รับคำสั่งเปิด/ปิดวาล์ว เช่น: farm/valve/1 → "1" หรือ "0"
  String msg = String((char*)payload).substring(0, len);
  String t   = String(topic);
  int    pin  = t.endsWith("1") ? VALVE1_PIN :
               t.endsWith("2") ? VALVE2_PIN :
               t.endsWith("3") ? VALVE3_PIN : VALVE4_PIN;
  digitalWrite(pin, msg == "1" ? LOW : HIGH);  // LOW = เปิด (Active-LOW)
}

void reconnectMQTT() {
  while (!mqtt.connected()) {
    if (mqtt.connect("SmartFarmESP32")) {
      mqtt.subscribe("farm/valve/#");
    } else {
      delay(3000);
    }
  }
}
Checklist ก่อนเปิดระบบ
ต่อ VCC Sensor ทั้งหมดที่ 3.3V (ไม่ใช่ 5V)
ต่อ Pull-up 10kΩ ที่ DHT22 DATA → 3.3V
ต่อ Pull-up 4.7kΩ ที่ DS18B20 DATA → 3.3V
Relay Module ต่อ VCC ที่ 5V (ไม่ใช่ 3.3V)
GND ของ ESP32 และ Relay ต่อร่วมกัน (Common GND)
ต่อ Valve ที่ขา NO (Normally Open) ของ Relay
ทดสอบ WiFi เชื่อมต่อสำเร็จ (Serial Monitor)
ค่า Soil Sensor อยู่ในช่วง 0–100% (Calibrate แล้ว)
DHT22 อ่านค่าอุณหภูมิและความชื้นได้ปกติ
DS18B20 อ่านค่าอุณหภูมิดินได้ (ไม่ใช่ -127°C)
Relay ส่งเสียงคลิกเมื่อรับคำสั่ง — ไฟ LED ติด
วาล์วเปิด-ปิดน้ำได้ปกติเมื่อรับคำสั่งจาก MQTT
Dashboard แสดงค่าเซนเซอร์ Realtime ได้
ระบบ Auto-irrigation ทำงาน (เปิด/ปิดตาม Threshold)
